0%

`); }); /* ======================================== MERGE API ======================================== */ app.post( '/merge', upload.array('files'), async(req,res)=>{ try{ const outputFile = path.join( __dirname, 'output', 'MERGED_MAINBILL.txt' ); /* WRITE STREAM */ const writer = fs.createWriteStream( outputFile, { encoding:'utf8' } ); /* HEADER 1X */ writer.write( HEADER + '\r\n' ); /* LOOP FILE */ for(const file of req.files){ console.log( 'PROCESS:', file.originalname ); /* READ FILE */ const data = fs.readFileSync( file.path, 'utf8' ); /* SPLIT */ const lines = data.split(/\r?\n/); /* LOOP BARIS */ for(let line of lines){ line = line.trim(); if(!line){ continue; } /* SKIP HEADER */ if( line .replace(/\s/g,'') .toUpperCase() === HEADER .replace(/\s/g,'') .toUpperCase() ){ continue; } writer.write( line + '\r\n' ); } /* DELETE TEMP */ fs.unlinkSync( file.path ); } /* FINISH */ writer.end(); writer.on('finish', ()=>{ console.log( 'MERGE SUCCESS' ); res.download( outputFile ); }); } catch(err){ console.log(err); res.status(500) .send('ERROR'); } }); /* ======================================== START SERVER ======================================== */ app.listen(PORT, ()=>{ console.log( 'SERVER RUNNING:', 'http://localhost:' + PORT ); });
